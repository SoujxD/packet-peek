cmake_minimum_required(VERSION 3.16)
project(libpeek VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_executable(libpeek src/main.c)
include(CTest)
if(BUILD_TESTING)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CRITERION REQUIRED IMPORTED_TARGET criterion)
  add_executable(peek_tests tests/test_smoke.c)
  target_link_libraries(peek_tests PRIVATE PkgConfig::CRITERION)
  add_test(NAME peek_tests COMMAND peek_tests)
endif()
